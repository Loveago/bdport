1. DNS CONFIGURATION (BOTH DOMAINS POINT TO SAME VPS IP)
================================================================================
A VPS has one public IP address. Both domains will point to that SAME IP.
When a visitor visits cmtshop.net, Nginx proxies to Next.js with Host: cmtshop.net,
serving customer storefronts. When a user visits commontech.net, they access the portal.

In your DNS manager (Namecheap, Cloudflare, GoDaddy, Hostinger, etc.):

For commontech.net:
| Type | Host / Name | Value / Destination | TTL |
| :--- | :--- | :--- | :--- |
| A    | @           | <YOUR_VPS_IP>       | Auto / 1 min |
| A    | www         | <YOUR_VPS_IP>       | Auto / 1 min |

For cmtshop.net:
| Type | Host / Name | Value / Destination | TTL |
| :--- | :--- | :--- | :--- |
| A    | @           | <YOUR_VPS_IP>       | Auto / 1 min |
| A    | www         | <YOUR_VPS_IP>       | Auto / 1 min |

(Both domains live together on the same server and database!)

2. STEP 1: CONNECT TO VPS & CONFIGURE FIREWALL
================================================================================
# SSH into your VPS (replace your_vps_ip with your actual VPS IP address)
ssh root@your_vps_ip

# Update system packages and install prerequisites
sudo apt update && sudo apt upgrade -y
sudo apt install -y curl git ufw build-essential libpq-dev nginx certbot python3-certbot-nginx

# Configure UFW Firewall (allows SSH, HTTP port 80, HTTPS port 443)
sudo ufw allow OpenSSH
sudo ufw allow 'Nginx Full'
sudo ufw --force enable
sudo ufw status

4. STEP 3: INSTALL & CONFIGURE POSTGRESQL
================================================================================
# Install PostgreSQL and contrib tools
sudo apt install -y postgresql postgresql-contrib

# Start and enable PostgreSQL service
sudo systemctl start postgresql
sudo systemctl enable postgresql

# Create database user and database
# (Replace 'YourSecurePassword123!' with your chosen secure database password)
sudo -u postgres psql << 'EOF'
CREATE USER commontech_user WITH PASSWORD 'YourSecurePassword123crazy!';
CREATE DATABASE commontech_db OWNER commontech_user;
GRANT ALL PRIVILEGES ON DATABASE commontech_db TO commontech_user;
ALTER USER commontech_user CREATEDB;
\q
EOF

# Grant schema privileges (PostgreSQL 15+)
sudo -u postgres psql -d commontech_db -c "GRANT ALL ON SCHEMA public TO commontech_user;"

# Test connecting to the local database
PGPASSWORD='YourSecurePassword123crazy!' psql -U commontech_user -d commontech_db -h 127.0.0.1 -c "\conninfo"

8. STEP 7: CONFIGURE PM2 PROCESS MANAGER
================================================================================
# Create PM2 ecosystem file
cat << 'EOF' > /var/www/commontech/ecosystem.config.js
module.exports = {
  apps: [
    {
      name: "commontech",
      cwd: "/var/www/commontech",
      script: "node_modules/next/dist/bin/next",
      args: "start -p 3000",
      instances: "max",
      exec_mode: "cluster",
      autorestart: true,
      watch: false,
      max_memory_restart: "1G",
      env: {
        NODE_ENV: "production",
        PORT: 3000
      }
    }
  ]
};
EOF

# Start application with PM2
pm2 start /var/www/commontech/ecosystem.config.js

# Save PM2 process list to persist across reboots
pm2 save

# Verify application responds locally
curl -I http://127.0.0.1:3000

9. STEP 8: CONFIGURE NGINX FOR BOTH DOMAINS
================================================================================
We create two modular Nginx site configs: one for commontech.net and one for cmtshop.net.

--- [ 8A: NGINX CONFIG FOR COMMONTECH.NET ] ---
cat << 'EOF' > /etc/nginx/sites-available/commontech.net
server {
    listen 80;
    listen [::]:80;
    server_name commontech.net www.commontech.net;

    client_max_body_size 250M;

    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;

        # WebSocket support
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';

        # Multi-tenant proxy headers
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;

        proxy_cache_bypass $http_upgrade;
        proxy_read_timeout 120s;
        proxy_connect_timeout 60s;
    }

    # Static file caching
    location /_next/static/ {
        alias /var/www/commontech/.next/static/;
        expires 365d;
        access_log off;
    }
}
EOF

--- [ 8B: NGINX CONFIG FOR CMTSHOP.NET ] ---
cat << 'EOF' > /etc/nginx/sites-available/cmtshop.net
server {
    listen 80;
    listen [::]:80;
    server_name cmtshop.net www.cmtshop.net;

    client_max_body_size 250M;

    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;

        # WebSocket support
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';

        # Multi-tenant proxy headers
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;

        proxy_cache_bypass $http_upgrade;
        proxy_read_timeout 120s;
        proxy_connect_timeout 60s;
    }

    # Static file caching
    location /_next/static/ {
        alias /var/www/commontech/.next/static/;
        expires 365d;
        access_log off;
    }
}
EOF

--- [ 8C: ENABLE SITES & TEST NGINX ] ---
sudo ln -sf /etc/nginx/sites-available/commontech.net /etc/nginx/sites-enabled/
sudo ln -sf /etc/nginx/sites-available/cmtshop.net /etc/nginx/sites-enabled/
sudo rm -f /etc/nginx/sites-enabled/default

# Test Nginx syntax
sudo nginx -t

# Reload Nginx
sudo systemctl reload nginx

10. STEP 9: ISSUE FREE SSL CERTIFICATES (CERTBOT HTTPS)
================================================================================

sudo certbot --nginx -d commontech.net -d www.commontech.net -d cmtshop.net -d www.cmtshop.net --non-interactive --agree-tos -m admin@commontech.net --redirect

# Test automatic SSL certificate renewal
sudo certbot renew --dry-run
